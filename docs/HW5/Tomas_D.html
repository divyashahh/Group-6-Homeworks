# building function
Query_News <- function(search, from_date, api_key) {
  response <- GET(
    url = "https://newsapi.org/v2/everything",
    query = list(
      q = search,
      from = from_date,
      language = "en",
      pageSize = 30,
      apikey = api_key
    )
  )
}
