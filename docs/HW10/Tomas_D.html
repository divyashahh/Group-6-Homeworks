set.seed(123)

# 75/25 split by season

# 10-fold CV split 
bike_folds <- vfold_cv(train_data, v = 10, strata = seasons)

Create MLR

# 1st MLR
bike_rec1 <- recipe(rented_bike_count ~ ., data = train_data) |>
  update_role(date, new_role = "ID") |>
  step_date(date, features = "dow") |>
  step_mutate(
    day_type = factor(if_else(date_dow %in% c("Sat", "Sun"), "Weekend", "Weekday"))
  ) |>
  step_rm(date_dow) |>
  step_normalize(all_numeric_predictors()) |>
  step_dummy(seasons, holiday, day_type)

Second MLR - First MLR + interaction

# 2nd MLR, 1 + interactions
bike_rec2 <- bike_rec1 |>
  step_interact(
    terms = ~ starts_with("seasons_"):starts_with("holiday_") +
      starts_with("seasons_"):temperature +
      temperature:rainfall
  )

# 3rd MLR: MLR 2 + quadratic terms for temperature and humidity
bike_rec3 <- bike_rec2 |>
  step_mutate(temp_sq = temperature^2, hum_sq = humidity^2)
