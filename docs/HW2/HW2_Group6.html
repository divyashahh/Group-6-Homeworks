Task 4: Control Flow Practice

For this last section, we added a status column to each data frame inside the list, then used a for loop with if/else logic to classify each patient as optimal, borderline, or high based on their post treatment/placebo blood pressure values.

# Create empty status columns in the data frames stored in the list
bp_list$treatment$status <- character(nrow(bp_list$treatment))
bp_list$placebo$status <- character(nrow(bp_list$placebo))

# Other method
bp_list$treatment$status <- character(20)
bp_list$placebo$status <- character(10)
for (i in 1:nrow(experiment_list$treatment)) {
  bp_value <- experiment_list$treatment$post_bp[i]
  if (bp_value <= 120) {
    experiment_list$treatment$status[i] <- "optimal"
  } else if (bp_value > 120 & bp_value <= 130) {
    experiment_list$treatment$status[i] <- "borderline"
  } else {
    experiment_list$treatment$status[i] <- "high"
  }
}
