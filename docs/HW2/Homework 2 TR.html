Task 4 Control Flow Practice

Loop logic for creating new columns for categories for non-placebo data

# Creating new character columns for categories 
list_bp$treatment$status <- character(20)

list_bp$placebo$status <- character(10)

# For loop logic for categorizing bp levels
for(i in 1:nrow(list_bp$treatment)){
  
  if (list_bp$treatment$post_bp[i] <= 120) {list_bp$treatment$status[i] <- "optimal"}
  
  else if (list_bp$treatment$post_bp[i] > 120 & list_bp$treatment$post_bp[i] <=130) {list_bp$treatment$status[i] <- "borderline"}
  
  else {list_bp$treatment$status[i] <- "high"}
}
